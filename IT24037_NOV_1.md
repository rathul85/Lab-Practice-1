package org.example;

import java.sql.*;
import java.util.Scanner;

public class Main {

    static final String URL  = "jdbc:postgresql://localhost:5432/prime_db";
    static final String USER = "postgres";
    static final String PASS = "12345678";   // <-- tomar asol PostgreSQL password

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            System.out.println("\n===== MENU =====");
            System.out.println("1. N-th prime insert koro");
            System.out.println("2. Database er sob data dekho");
            System.out.println("3. Exit");
            System.out.print("Choice: ");
            int choice = sc.nextInt();

            switch (choice) {
                case 1:
                    System.out.print("N koto? ");
                    int n = sc.nextInt();
                    if (n <= 0) {
                        System.out.println("N must be 1 or more!");
                        break;
                    }
                    int prime = findNthPrime(n);
                    System.out.println(n + "-th prime = " + prime);
                    insertToDB(n, prime);
                    break;
                case 2:
                    showAll();
                    break;
                case 3:
                    System.out.println("Bye!");
                    return;
                default:
                    System.out.println("Invalid choice!");
            }
        }
    }

    static int findNthPrime(int n) {
        int count = 0;
        int num = 1;
        while (count < n) {
            num++;
            if (isPrime(num)) {
                count++;
            }
        }
        return num;
    }

    static boolean isPrime(int num) {
        if (num < 2) return false;
        for (int i = 2; i * i <= num; i++) {
            if (num % i == 0) return false;
        }
        return true;
    }

    static void insertToDB(int n, int prime) {
        String sql = "INSERT INTO nth_prime (n, prime_number) VALUES (?, ?)";

        try (Connection con = DriverManager.getConnection(URL, USER, PASS);
             PreparedStatement ps = con.prepareStatement(sql)) {

            ps.setInt(1, n);
            ps.setInt(2, prime);
            int rows = ps.executeUpdate();

            if (rows > 0) {
                System.out.println("Database e save hoyeche!");
            }
        } catch (SQLException e) {
            System.out.println("DB Error: " + e.getMessage());
        }
    }

    static void showAll() {
        String sql = "SELECT * FROM nth_prime ORDER BY id";

        try (Connection con = DriverManager.getConnection(URL, USER, PASS);
             Statement st = con.createStatement();
             ResultSet rs = st.executeQuery(sql)) {

            System.out.println("\nID\tN\tPrime");
            System.out.println("------------------------");
            while (rs.next()) {
                System.out.println(rs.getInt("id") + "\t"
                        + rs.getInt("n") + "\t"
                        + rs.getInt("prime_number"));
            }
        } catch (SQLException e) {
            System.out.println("DB Error: " + e.getMessage());
        }
    }
}
